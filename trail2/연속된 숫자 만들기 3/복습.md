## **📝문제 설명**
3명의 사람이 일직선 위에 서있습니다.<br/> 
세 사람이 서있는 위치가 주어지면 양 끝쪽에 있는 사람 중 한 사람을 선택해 항상 남은 두 사람 사이로 이동시켜 결국 세 사람이 서 있는 위치가 연속된 숫자가 되도록 만들려고 합니다.<br/> 
연속된 수가 되기 위한 최대 이동 횟수를 구해 출력하는 프로그램을 작성해보세요.<br/>

### **🧾나의 풀이**

```js
const fs = require("fs");
const input = fs.readFileSync(0).toString().trim().split('\n');
const a = input[0].split(" ").map(Number);
// Please Write your code here.

// 최대 이동 횟수를 구해야 해
let move = 0
let longer = 0
let place = [0, 0]

while (true) {
    let gap1 = a[1] - a[0] 
    let gap2 = a[2] - a[1]

    // 이미 연속된 수라면 이동거리를 출력한다.
    if (gap1 === 1 && gap2 === 1) {
        console.log(move)
        break;

    } else {
        // 0, 1 번 사이의 거리과 1,2번 사이의 거리 중에서 더 긴 것을 선택한다
        if (gap1 > gap2) {
            longer = gap1;
            place = [a[0], a[1]];
            // 다른 한 쪽 끝 수를 더 긴 쌍의 중간값으로 이동한다
            a[2] = a[1]
            a[1] = a[0] + Math.floor(gap1 / 2)

            move++

        } else if (gap1 < gap2) {
            longer = gap2;
            place = [a[1], a[2]];

            // 다른 한 쪽 끝 수를 더 긴 쌍의 중간값으로 이동한다
            a[0] = a[1]
            a[1] = a[0] + Math.floor(gap2 / 2)
            move++
        }

    }
}

console.log(move)
```
